cmake_minimum_required(VERSION 3.22)
project(PaldoYadam VERSION 0.8.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(PALDO_BUILD_GAME "Build the native graphical game" ON)
option(PALDO_BUILD_TESTS "Build deterministic gameplay tests" ON)
option(PALDO_SANITIZE "Enable address/undefined behavior checks" OFF)

add_library(paldo_core STATIC src/game.c src/world.c src/maps.c src/save.c src/save_store.c)
set_target_properties(paldo_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(paldo_core PUBLIC src)
if(NOT MSVC)
    target_link_libraries(paldo_core PUBLIC m)
    target_compile_options(paldo_core PRIVATE -Wall -Wextra -Wpedantic)
else()
    target_compile_options(paldo_core PUBLIC /utf-8)
endif()
if(PALDO_SANITIZE AND NOT MSVC AND NOT ANDROID)
    target_compile_options(paldo_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(paldo_core PUBLIC -fsanitize=address,undefined)
endif()

if(PALDO_BUILD_GAME)
    if(ANDROID)
        if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/SDL/CMakeLists.txt")
            message(FATAL_ERROR "Run python3 scripts/prepare_android.py before building Android.")
        endif()
        set(SDL_SHARED ON CACHE BOOL "" FORCE)
        set(SDL_STATIC OFF CACHE BOOL "" FORCE)
        set(SDL_TEST OFF CACHE BOOL "" FORCE)
        set(SDL_TESTS OFF CACHE BOOL "" FORCE)
        add_subdirectory(third_party/SDL EXCLUDE_FROM_ALL)
        target_link_options(SDL2 PRIVATE "-Wl,-z,max-page-size=16384")
    elseif(PALDO_SDL_INCLUDE_DIR AND PALDO_SDL_LIBRARY)
        # Allows verification against an already installed SDL runtime without changing the OS.
        add_library(SDL2::SDL2 UNKNOWN IMPORTED)
        set_target_properties(SDL2::SDL2 PROPERTIES IMPORTED_LOCATION "${PALDO_SDL_LIBRARY}"
            INTERFACE_INCLUDE_DIRECTORIES "${PALDO_SDL_INCLUDE_DIR}")
    else()
        find_package(SDL2 2.0.18 CONFIG QUIET)
        if(NOT TARGET SDL2::SDL2)
            find_package(PkgConfig REQUIRED)
            pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.18)
            add_library(SDL2::SDL2 ALIAS PkgConfig::SDL2)
        endif()
    endif()

    if(ANDROID)
        add_library(main SHARED src/main.c src/draw.c)
        set(PALDO_TARGET main)
        target_link_options(main PRIVATE "-Wl,-z,max-page-size=16384")
    else()
        add_executable(paldo-yadam src/main.c src/draw.c)
        set(PALDO_TARGET paldo-yadam)
        if(TARGET SDL2::SDL2main)
            target_link_libraries(paldo-yadam PRIVATE SDL2::SDL2main)
        endif()
    endif()
    target_include_directories(${PALDO_TARGET} SYSTEM PRIVATE third_party/stb)
    target_link_libraries(${PALDO_TARGET} PRIVATE paldo_core SDL2::SDL2)
    if(MSVC)
        target_compile_options(${PALDO_TARGET} PRIVATE /utf-8)
    else()
        target_compile_options(${PALDO_TARGET} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(NOT ANDROID)
        add_custom_command(TARGET ${PALDO_TARGET} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:${PALDO_TARGET}>/assets")
        install(TARGETS ${PALDO_TARGET} RUNTIME DESTINATION .)
        install(DIRECTORY assets DESTINATION .)
    endif()
endif()
